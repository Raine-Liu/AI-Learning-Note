```python
class InceptionA(nn.Module):
	def __init__(self, in_channel):
		super(InceptionA, self).__init__()
		self.branch1x1 = nn.Conv2d(in_channels, 16, kernel_size = 1)
		self.branch5x5_1 = nn.Conv2d(in_channels, 16, kernel_size=1)
		self.branch5x5_2 = nn.Conv2d(16, 24, kernel_size=5, padding=2)
		self.branch3x3_1 = nn.Conv2d(in_channels, 16, kernel_size=1)
		self.branch3x3_2 = nn.Conv2d(16, 24, kernel_size=3, padding=1)
		self.branch3x3_2 = nn.Conv2d(24, 24, kernel_size=3, padding=1)
		self.branch_pool = nn.Conv2d(in_channels, 24, kernel_size = 1)
	def forward(self, x):
		branch1x1 = self.branch1x1(x)
		
		branch5x5 = self.branch5x5_1(x)
		branch5x5 = self.branch5x5_2(branch5x5)
		
		branch3x3 = self.branch3x3_1(x)
		branch3x3 = self.branch3x3_2(branch3x3)
		branch3x3 = self.branch3x3_3(branch3x3)
		branch_pool = F.avg_pool2d(x, kernel_size, stride=1, padding=1) #函数，forward参数；池化分支
		branch_pool = self.branch_pool(branch_pool)
		
		outputs = [branch1x1, branch5x5, branch3x3, branch_pool] #输出88个通道
		return torch.cat(outputs, dim=1)
```

#### Residual net
Plain nets: stacking 3\*3 conv layers
叠更多的3\*3,效果反而变差——梯度消失，浅层没有得到充分训练
为改善这种情况，将前面的层锁住，逐层训练——但是深度学习的层数太多了，这种操作十分麻烦
###### Deep Residual Learning
![10. Advanced CNN-1783763914600.webp](images/10.%20Advanced%20CNN-1783763914600.webp)
增加了跳连接
$$令z=H(x),\frac{\partial z}{\partial x}=\frac{\partial F}{\partial x}+1$$

偏导在1附近，在梯度很小的时候多个相乘可以解决梯度消失的问题

residual block
```python
class ResidualBlock(nn.Module):
	def __init__(self, channels):
		super(ResidualBlock, self).__init__()
		self.channels = channels
		self.conv1 = nn.Conv2d(channels, channels, kernel_size=3,padding=1) #通过补零保证输入输出尺寸一致
		self.conv2 = nn.Conv2d(channels, channels, kernel_size=3,padding=1)
	
	def forward(self, x):
		y = F.relu(self.conv1(x))
		y = self.conv2(y)
		return F.relu(x + y) #F(x)+x
```

```python
class Net(nn.Module):
	def __init__(self):
		self.conv1 = nn.Conv2d(1, 16, kernel_size=5)
		self.conv2 = nn.Conv2d(16, 32, kernel_size=5)
		self.mp = nn.MaxPool2d(2)
		
		self.rblock1 = ResidualBlock(16)
		self.rblock2 = ResidualBlock(32)
		
		self.fc = nn.Linear(512, 10)
	
	def forward(self, x):
		in_size = x.size(0)
		x = self.mp(F.relu(self.conv1(x)))
		x = self.rblock1(x)
		x = self.mp(F.relu(self.conv2(x)))
		x = self.rblock2(x)
		x = x.view(in_size, -1)
		x = self.fc(x)
		return x
```
