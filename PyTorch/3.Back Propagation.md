#### Tensor in PyTorch
**Tensor**：contains **Data** and **Grad** (value and gradient)（grad 也是tensor)
```python
import torch

x_data = [1.0,2.0,3.0]
y_data = [2.0,4.0,6.0]

w = torch.Tensor([1.0])
w.requires_grad = True #设定为需要计算梯度

def forward(x):
	return x * w #发生了自动类型转换，将x变为tensor

def loss(x,y):
	y_pred = forward(x)
	return (y_pred - y) ** 2 
#构建计算图(使用张量）

print("predict (before training)", 4, forward(4).item())

for epoch in range(100):
	for x, y in zip(x_data, y_data):
		l = loss(x, y) #forward计算loss
		l.backward() #Backward 把计算链路上所有需要梯度的地方计算出来；backward后计算图释放
		print('\tgrad:', x, y, w.grad.item())
		w.data =w.data - 0.01 * w.grad.data #取grad 的data进行计算，不会生成计算图
		w.grad.data.zero_() #更新后将w中的梯度全部清零

	print("progress:", epoch, l.item())
print("predict (after training)", 4, forward(4).item())
```

如果要计算所有的loss，记得.item`sum+=l.item()` 否则sum为数，l为张量，是计算图，多轮迭代可能吃光内存

现代写法：
```python
import torch

x_data = [1.0, 2.0, 3.0]
y_data = [2.0, 4.0, 6.0]

w = torch.Tensor([1.0])
w.requires_grad = True

def forward(x):
	return x * w

def loss(x, y):
	y_pred = forward(x)
	return (y_pred - y) ** 2

print("predict (before training)", 4, forward(4).item())

for epoch in range(100):
	for x, y in zip(x_data, y_data):
		l = loss(x, y)

		l.backward()

		print('\tgrad:', x, y, w.grad.item())

		with torch.no_grad():
			w -= 0.01 * w.grad

		w.grad.zero_()

	print("progress:", epoch, l.item())

print("predict (after training)", 4, forward(4).item())
```
